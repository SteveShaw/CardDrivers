cmake_minimum_required(VERSION 3.20)
project(roudi_ipc LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This project requires Linux")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(ipc src/ipc.cpp)
target_include_directories(ipc PUBLIC include)
target_compile_definitions(ipc PUBLIC _GNU_SOURCE)
target_compile_options(ipc PUBLIC -Wall -Wextra -Wpedantic -Werror)
target_link_libraries(ipc PUBLIC rt)
add_executable(producer src/producer.cpp)
target_link_libraries(producer PRIVATE ipc)
add_executable(consumer src/consumer.cpp)
find_package(Threads REQUIRED)
target_link_libraries(consumer PRIVATE ipc Threads::Threads)
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME smoke COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/smoke.py $<TARGET_FILE_DIR:producer>)
set_tests_properties(smoke PROPERTIES TIMEOUT 45 RUN_SERIAL TRUE)
